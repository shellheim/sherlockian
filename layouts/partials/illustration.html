{{- $page := .page -}} {{- $caption := .caption -}} {{- $number := .number | default 1 -}} {{- $base := "" -}} {{- if isset $page.Params "chapter" -}} {{- $base = printf "images/illustrations/%s/chapter-%d" ($page.Params.novel | urlize) $page.Params.chapter -}} {{- else -}} {{- $base = printf "images/illustrations/%s" ($page.Params.title | urlize) -}} {{- end -}} {{- if gt $number 1 -}} {{- $suffixes := slice "two" "three" "four" "five" "six" "seven" "eight" "nine" "ten" -}} {{- $suffix := index $suffixes (sub $number 2) -}} {{- $base = printf "%s-%s" $base $suffix -}} {{- end -}} {{- $img := resources.Get (printf "%s.jpg" $base) -}} {{- if not $img -}} {{- $img = resources.Get (printf "%s.png" $base) -}} {{- end -}} {{- with $img -}} {{- $maxW := 1400 -}} {{- $w := $maxW -}} {{- if lt .Width $maxW -}} {{- $w = .Width -}} {{- end -}} {{- $full := .Resize (printf "%dx webp q80" $w) -}} {{- $small := .Resize (printf "%dx webp q80" (int (div $w 2))) -}} {{- end -}}
